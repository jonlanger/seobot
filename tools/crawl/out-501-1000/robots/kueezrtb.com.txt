  document.head.appendChild(styleSheet);

  // ====================================================================
  // Part 2: The main logic of the carousel
  // ====================================================================
  const wrapper = document.querySelector('.wrapper_logos_hide_margins');
  const logosContainer = document.querySelector('.logos-clients-images');

  if (!wrapper || !logosContainer) {
    console.warn("Carousel elements not found. Check classes: .wrapper_logos_hide_margins, .logos-clients-images");
    return;
  }

  const originalItemsHTML = logosContainer.innerHTML;
  let isAnimating = false;

  const setupCarousel = () => {
    // We wait for one frame so the browser correctly calculates widths after reset
    requestAnimationFrame(() => {
      const wrapperWidth = wrapper.offsetWidth;
      // Measure the width of the ORIGINAL set of logos
      const originalLogosWidth = logosContainer.scrollWidth;

      // --- CONDITION: Start the animation only if the logos don’t fit ---
      if (originalLogosWidth > wrapperWidth) {
        if (isAnimating) return; // If already animating, do nothing
        isAnimating = true;

        // --- CLONING: Duplicate the content to fill the gap ---
        const originalItems = Array.from(logosContainer.children);
        originalItems.forEach(item => {
          const clone = item.cloneNode(true);
          clone.setAttribute('aria-hidden', 'true');
          logosContainer.appendChild(clone);
        });

        // --- CORRECT ANIMATION ---
        // 1. Define the exact movement distance
        const moveDistance = originalLogosWidth;

        // 2. Calculate a proportional duration for constant speed
        // You can change the number '40' to make the animation faster or slower
        const duration = moveDistance / 40;

        // 3. Create and inject the @keyframes animation rule
        const animationName = `scroll_left_${Date.now()}`;
        const keyframes = `
          @keyframes ${animationName} {
            from {
              transform: translateX(0);
            }
            to {
              transform: translateX(-${moveDistance}px);
            }
          }
        `;
        styleSheet.innerHTML = keyframes;

        // 4. Apply the animation to the element
        logosContainer.style.animation = `${animationName} ${duration}s linear infinite`;

      } else {
        // --- STOP: If the logos fit, stop everything ---
        if (!isAnimating) return;
        isAnimating = false;
        logosContainer.style.animation = '';
        logosContainer.innerHTML = originalItemsHTML; // Reset to original state
      }
    });
  };

  // Use ResizeObserver to automatically recalculate everything on resize
  const observer = new ResizeObserver(() => {
     // Completely reset state before recalculating
     isAnimating = false;
     logosContainer.style.animation = '';
     logosContainer.innerHTML = originalItemsHTML;
     setupCarousel();
  });

  observer.observe(wrapper);

});
